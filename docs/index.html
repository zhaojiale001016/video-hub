Quickstart

跑起来

bash · Python ≥3.11 + ffmpeg · biliup 见 README复制
# 环境
python3 -m venv .venv && .venv/bin/pip install -r requirements.txt

# 配置（首启自动从 example 复制）
cp config/sources.example.yaml config/sources.yaml

# 扫码登录(B站 + 任意 sau 平台)
.venv/bin/python3 cli.py login && .venv/bin/python3 cli.py login-ks && .venv/bin/python3 cli.py login-sph
# 更多平台: cli.py login-extra douyin

# 启动 → http://127.0.0.1:8600
.venv/bin/python3 cli.py serve
